cmake_minimum_required(VERSION 3.24)
project(v8_cmake VERSION 0.1.0 LANGUAGES CXX)

if(TARGET v8::v8)
  return()
endif()

find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
include(cmake/V8Platform.cmake)
v8_cmake_platform(_v8_os _v8_cpu)

option(V8_BUILD_EXAMPLE "Build the embedding smoke test" ${PROJECT_IS_TOP_LEVEL})
option(V8_BUILD_TESTS "Register wrapper and embedding tests" ${PROJECT_IS_TOP_LEVEL})
option(V8_ENABLE_I18N "Embed ICU and enable JavaScript Intl" ON)
option(V8_OFFLINE "Forbid downloads, syncs and dependency hooks" OFF)
set(V8_SOURCE_DIR "" CACHE PATH "Optional, already synced V8 checkout (not modified)")
set(V8_DEPOT_TOOLS_DIR "" CACHE PATH "Optional, already installed depot_tools")
set(V8_WORKSPACE "${CMAKE_CURRENT_BINARY_DIR}/workspace" CACHE PATH "Managed checkout and dependency cache")
set(V8_BUILD_JOBS "" CACHE STRING "Ninja parallelism; empty uses Ninja's default")
set(V8_GN_EXECUTABLE "" CACHE FILEPATH "Override the GN from V8 DEPS")
set(V8_NINJA_EXECUTABLE "" CACHE FILEPATH "Override the Ninja from V8 DEPS")

if(V8_BUILD_JOBS AND NOT V8_BUILD_JOBS MATCHES "^[1-9][0-9]*$")
  message(FATAL_ERROR "V8_BUILD_JOBS must be a positive integer")
endif()

set(_v8_configs Debug Release RelWithDebInfo MinSizeRel)
foreach(config IN LISTS CMAKE_CONFIGURATION_TYPES)
  if(NOT config IN_LIST _v8_configs)
    message(FATAL_ERROR "V8: unsupported configuration '${config}'")
  endif()
endforeach()
if(CMAKE_BUILD_TYPE AND NOT CMAKE_BUILD_TYPE IN_LIST _v8_configs)
  message(FATAL_ERROR "V8: unsupported CMAKE_BUILD_TYPE '${CMAKE_BUILD_TYPE}'")
endif()
set(_v8_config "$<IF:$<BOOL:$<CONFIG>>,$<CONFIG>,Release>")
set(_v8_stage "${CMAKE_CURRENT_BINARY_DIR}/stage")
set(_v8_runtime "${CMAKE_MSVC_RUNTIME_LIBRARY}")
if(NOT _v8_runtime)
  set(_v8_runtime "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
endif()
set(_v8_deployment "${CMAKE_OSX_DEPLOYMENT_TARGET}")
if(_v8_os STREQUAL "mac")
  if(NOT _v8_deployment)
    set(_v8_deployment "12.0")
  elseif(_v8_deployment VERSION_LESS "12.0")
    message(FATAL_ERROR "V8 requires a macOS deployment target of at least 12.0")
  endif()
endif()

# JSON is generated per configuration, including on Visual Studio and Xcode.
set(_v8_json "{\n")
macro(v8_json key value)
  set(_value "${value}")
  string(REPLACE "\\" "\\\\" _value "${_value}")
  string(REPLACE "\"" "\\\"" _value "${_value}")
  string(REPLACE "\n" "\\n" _value "${_value}")
  string(APPEND _v8_json "  \"${key}\": \"${_value}\",\n")
endmacro()
v8_json(lock "${CMAKE_CURRENT_SOURCE_DIR}/v8-version.json")
v8_json(patch "${CMAKE_CURRENT_SOURCE_DIR}/patches/windows-runtime.patch")
v8_json(workspace "${V8_WORKSPACE}")
v8_json(source "${V8_SOURCE_DIR}")
v8_json(depot_tools "${V8_DEPOT_TOOLS_DIR}")
v8_json(output "${CMAKE_CURRENT_BINARY_DIR}/gn/${_v8_config}")
v8_json(stage "${_v8_stage}/${_v8_config}")
v8_json(config "${_v8_config}")
v8_json(target_os "${_v8_os}")
v8_json(target_cpu "${_v8_cpu}")
v8_json(runtime "${_v8_runtime}")
v8_json(deployment_target "${_v8_deployment}")
v8_json(sysroot "${CMAKE_SYSROOT}")
v8_json(mac_sdk "${CMAKE_OSX_SYSROOT}")
v8_json(i18n "${V8_ENABLE_I18N}")
v8_json(offline "${V8_OFFLINE}")
v8_json(jobs "${V8_BUILD_JOBS}")
v8_json(gn "${V8_GN_EXECUTABLE}")
v8_json(ninja "${V8_NINJA_EXECUTABLE}")
string(APPEND _v8_json "  \"schema\": 1\n}\n")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/config/${_v8_config}.json" CONTENT "${_v8_json}")

if(WIN32)
  set(_v8_library v8_monolith.lib)
else()
  set(_v8_library libv8_monolith.a)
endif()
# These directories must exist at generate time; headers arrive at build time.
foreach(config IN LISTS _v8_configs)
  file(MAKE_DIRECTORY "${_v8_stage}/${config}/include")
endforeach()
add_custom_target(v8_build
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/build_v8.py"
    --config "${CMAKE_CURRENT_BINARY_DIR}/config/${_v8_config}.json"
  BYPRODUCTS "${_v8_stage}/${_v8_config}/lib/${_v8_library}"
             "${_v8_stage}/${_v8_config}/include/v8-gn.h"
  USES_TERMINAL VERBATIM
  COMMENT "Building V8 with its pinned GN toolchain")
add_library(v8_monolith STATIC IMPORTED GLOBAL)
add_library(v8::v8 ALIAS v8_monolith)
add_library(v8::monolith ALIAS v8_monolith)
add_dependencies(v8_monolith v8_build)
set_target_properties(v8_monolith PROPERTIES
  IMPORTED_CONFIGURATIONS "${_v8_configs}"
  IMPORTED_LOCATION "${_v8_stage}/Release/lib/${_v8_library}"
  IMPORTED_LINK_INTERFACE_LANGUAGES CXX
  INTERFACE_INCLUDE_DIRECTORIES "${_v8_stage}/${_v8_config}/include"
  INTERFACE_COMPILE_FEATURES cxx_std_20
  INTERFACE_COMPILE_DEFINITIONS V8_GN_HEADER)
foreach(config IN LISTS _v8_configs)
  string(TOUPPER "${config}" upper)
  set_property(TARGET v8_monolith PROPERTY "IMPORTED_LOCATION_${upper}" "${_v8_stage}/${config}/lib/${_v8_library}")
endforeach()
target_link_libraries(v8_monolith INTERFACE Threads::Threads ${CMAKE_DL_LIBS})
if(_v8_os STREQUAL "linux")
  target_link_libraries(v8_monolith INTERFACE rt)
elseif(_v8_os STREQUAL "mac")
  target_link_libraries(v8_monolith INTERFACE
    "-framework CoreFoundation" "-framework Foundation" "-framework Security")
elseif(_v8_os STREQUAL "win")
  target_link_libraries(v8_monolith INTERFACE dbghelp winmm ws2_32 advapi32)
  target_compile_options(v8_monolith INTERFACE /Zc:__cplusplus)
endif()

if(V8_BUILD_TESTS)
  enable_testing()
  add_test(NAME v8_wrapper COMMAND "${Python3_EXECUTABLE}" -m unittest discover
    -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -v)
endif()

  if(V8_BUILD_TESTS)
    add_test(NAME v8_hello COMMAND v8_hello)

  endif()
endif()
