cmake_minimum_required(VERSION 3.24)
project(v8_fetchcontent_example LANGUAGES CXX)
include(FetchContent)
FetchContent_Declare(v8_wrapper SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../..")
FetchContent_MakeAvailable(v8_wrapper)
add_executable(hello ../hello.cc)
target_link_libraries(hello PRIVATE v8::v8)
enable_testing()
add_test(NAME embedding COMMAND hello)
set_tests_properties(embedding PROPERTIES PASS_REGULAR_EXPRESSION "V8 .*: 42")
